void AddMaterial(){
 if(!CanWrite())return;
 var a=MaterialPicker.Ask(this,stock);
 if(a==null)return;
 WriteAction("addMaterial",new Dictionary<string,object>{{"name",a[0]},{"group",a[1]},{"sub",a[2]},{"spec",a[3]},{"unit",a[4]},{"brand",a[5]}});
}

// Add this class outside MainForm; same namespace, references as existing client.
static class MaterialPicker {
 static bool Eq(string a,string b){return String.Equals((a??"").Trim(),(b??"").Trim(),StringComparison.CurrentCultureIgnoreCase);}
 static string[] Choices(IEnumerable<string> values){return values.Where(v=>!String.IsNullOrWhiteSpace(v)).Select(v=>v.Trim()).Distinct(StringComparer.CurrentCultureIgnoreCase).OrderBy(v=>v,StringComparer.CurrentCultureIgnoreCase).ToArray();}
 static void Fill(ComboBox c,IEnumerable<string> values){
  string old=c.Text;var items=Choices(values);c.BeginUpdate();
  try{c.AutoCompleteSource=AutoCompleteSource.None;c.Items.Clear();c.Items.AddRange(items);c.Text=old;c.AutoCompleteSource=AutoCompleteSource.ListItems;}finally{c.EndUpdate();}
 }
 public static string[] Ask(Form owner,List<string[]> stock){
  var rows=stock.Where(r=>r!=null&&r.Length>=7&&!String.IsNullOrWhiteSpace(r[1])).Select(r=>(string[])r.Clone()).ToList();
  using(var f=new Form{Text="Thêm vật tư mới",ClientSize=new Size(660,490),Font=new Font("Segoe UI",11),StartPosition=FormStartPosition.CenterParent,FormBorderStyle=FormBorderStyle.FixedDialog,MaximizeBox=false,MinimizeBox=false,BackColor=Color.White}){
   string[] labels={"Tên vật tư (chọn hoặc nhập mới)","Nhóm chính","Nhóm con (theo nhóm chính)","Thông số kỹ thuật mới","ĐVT","Hãng"};
   var c=new ComboBox[6];bool updating=false;
   for(int i=0;i<6;i++){int y=12+i*62;f.Controls.Add(new Label{Text=labels[i],Left=20,Top=y,Width=620,Height=24});c[i]=new ComboBox{DropDownStyle=ComboBoxStyle.DropDown,AutoCompleteMode=AutoCompleteMode.SuggestAppend,AutoCompleteSource=AutoCompleteSource.ListItems};c[i].SetBounds(20,y+24,620,30);f.Controls.Add(c[i]);}
   c[3].AutoCompleteMode=AutoCompleteMode.None;
   Fill(c[0],rows.Select(r=>r[1]));Fill(c[1],rows.Select(r=>r[3]));Fill(c[2],rows.Select(r=>r[4]));Fill(c[4],rows.Select(r=>r[5]).Concat(new[]{"Cái"}));Fill(c[5],rows.Select(r=>r[6]));c[4].Text="Cái";
   Action groups=()=>{if(updating)return;updating=true;try{var list=rows.Where(r=>Eq(r[3],c[1].Text)).ToList();if(!list.Any(r=>Eq(r[4],c[2].Text)))c[2].Text="";Fill(c[2],list.Select(r=>r[4]));}finally{updating=false;}};
   Action name=()=>{if(updating)return;updating=true;try{
    var matches=rows.Where(r=>Eq(r[1],c[0].Text)).ToList();if(matches.Count==0)return;
    var gs=Choices(matches.Select(r=>r[3]));if(gs.Length==1)c[1].Text=gs[0];else c[1].Text="";
    var ss=Choices(matches.Where(r=>Eq(r[3],c[1].Text)).Select(r=>r[4]));c[2].Text=ss.Length==1?ss[0]:"";
    Fill(c[2],rows.Where(r=>Eq(r[3],c[1].Text)).Select(r=>r[4]));
    var us=Choices(matches.Select(r=>r[5]));c[4].Text=us.Length==1?us[0]:"";
    var bs=Choices(matches.Select(r=>r[6]));c[5].Text=bs.Length==1?bs[0]:"";
    c[3].Text="";Fill(c[3],matches.Select(r=>r[2]));
   }finally{updating=false;}};
   c[0].SelectionChangeCommitted+=(s,e)=>name();c[0].Leave+=(s,e)=>{if(c[1].Text.Trim()==""&&c[2].Text.Trim()=="")name();};
   c[1].SelectionChangeCommitted+=(s,e)=>groups();c[1].Leave+=(s,e)=>{if(!updating){updating=true;try{Fill(c[2],rows.Where(r=>Eq(r[3],c[1].Text)).Select(r=>r[4]));}finally{updating=false;}}};
   var hint=new Label{Text="Chọn tên cũ để kế thừa nhóm; nhập quy cách mới. Tên + quy cách trùng sẽ bị từ chối.",Left=20,Top=390,Width=620,Height=36};f.Controls.Add(hint);
   string[] result=null;var ok=new Button{Text="Xác nhận",Left=400,Top=438,Width=115,Height=36,BackColor=Color.FromArgb(22,117,209),ForeColor=Color.White};var cancel=new Button{Text="Hủy",Left=525,Top=438,Width=115,Height=36,DialogResult=DialogResult.Cancel};
   ok.Click+=(s,e)=>{var a=c.Select(x=>x.Text.Trim()).ToArray();if(new[]{0,1,2,4}.Any(i=>a[i]=="")){MessageBox.Show("Nhập đủ tên, nhóm chính, nhóm con và ĐVT.");return;}if(rows.Any(r=>Eq(r[1],a[0])&&Eq(r[2],a[3]))){MessageBox.Show("Tên và thông số đã có. Hãy chọn vật tư hiện có hoặc nhập quy cách khác.");return;}result=a;f.DialogResult=DialogResult.OK;};
   f.Controls.AddRange(new Control[]{ok,cancel});f.AcceptButton=ok;f.CancelButton=cancel;return f.ShowDialog(owner)==DialogResult.OK?result:null;
  }
 }
